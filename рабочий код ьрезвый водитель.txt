import json
import urllib.request
import ssl
import time

TOKEN = "8099391152:AAG4UDErsqzn7cg7psJgcZEX_Hbb_5N8GcA"
ADMIN_ID = 7465925576
URL = f"https://api.telegram.org/bot{TOKEN}/"

ctx = ssl.create_default_context()
ctx.check_hostname = False
ctx.verify_mode = ssl.CERT_NONE

def api(method, data=None):
    if data:
        data = json.dumps(data).encode()
    req = urllib.request.Request(URL + method, data=data, headers={"Content-Type": "application/json"})
    with urllib.request.urlopen(req, context=ctx) as r:
        return json.loads(r.read().decode())

def send_message(chat_id, text, reply_markup=None):
    data = {"chat_id": chat_id, "text": text, "parse_mode": "HTML"}
    if reply_markup:
        data["reply_markup"] = reply_markup
    return api("sendMessage", data)

def get_updates(offset=None):
    return api("getUpdates", {"offset": offset, "timeout": 30})["result"]

def get_user_location(message):
    loc = message.get("location")
    if loc:
        lat = loc["latitude"]
        lon = loc["longitude"]
        name = message["from"].get("first_name", "Без имени")
